cmake_minimum_required(VERSION 3.24)
project(PrismPlayer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Sql Test)
add_library(prism_core src/core.cpp src/core.h src/library.cpp src/library.h)
target_link_libraries(prism_core PUBLIC Qt6::Core Qt6::Sql)
target_include_directories(prism_core PUBLIC src)
add_executable(PrismPlayer WIN32 src/main.cpp src/window.cpp src/window.h src/canvas.cpp src/canvas.h src/mpv_backend.cpp src/mpv_backend.h)
target_include_directories(PrismPlayer PRIVATE .deps/mpv/include)
target_link_libraries(PrismPlayer PRIVATE prism_core Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL)
if(MSVC)
  target_compile_options(prism_core PRIVATE /W4 /utf-8)
  target_compile_options(PrismPlayer PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(prism_tests tests/core_tests.cpp)
target_link_libraries(prism_tests PRIVATE prism_core Qt6::Test)
add_test(NAME prism_core COMMAND prism_tests)
add_executable(prism_backend_tests tests/backend_tests.cpp src/mpv_backend.cpp src/mpv_backend.h)
target_include_directories(prism_backend_tests PRIVATE src .deps/mpv/include)
target_link_libraries(prism_backend_tests PRIVATE Qt6::Core Qt6::Test)
add_test(NAME prism_backend COMMAND prism_backend_tests)
