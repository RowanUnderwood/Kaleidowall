cmake_minimum_required(VERSION 3.24)
project(Kaleidowall VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Sql Test)
add_library(kaleido_core src/core.cpp src/core.h src/library.cpp src/library.h)
target_link_libraries(kaleido_core PUBLIC Qt6::Core Qt6::Sql)
target_include_directories(kaleido_core PUBLIC src)
add_library(kaleido_export src/compositor.cpp src/compositor.h src/export_timeline.cpp src/export_timeline.h src/exporter.cpp src/exporter.h src/gpu.cpp src/gpu.h)
target_link_libraries(kaleido_export PUBLIC kaleido_core Qt6::Gui Qt6::OpenGL)
if(WIN32)
  target_link_libraries(kaleido_export PRIVATE dxgi)
endif()
add_executable(Kaleidowall WIN32 src/main.cpp src/window.cpp src/window.h src/window_export.cpp src/export_dialog.cpp src/export_dialog.h src/canvas.cpp src/canvas.h src/mpv_backend.cpp src/mpv_backend.h)
qt_add_resources(Kaleidowall app_icons
  PREFIX "/icons"
  BASE assets
  FILES assets/kaleidowall.ico
)
if(WIN32)
  enable_language(RC)
  target_sources(Kaleidowall PRIVATE assets/kaleidowall.rc)
endif()
target_include_directories(Kaleidowall PRIVATE .deps/mpv/include)
target_link_libraries(Kaleidowall PRIVATE kaleido_export Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL)
if(MSVC)
  target_compile_options(kaleido_core PRIVATE /W4 /utf-8)
  target_compile_options(kaleido_export PRIVATE /W4 /utf-8)
  target_compile_options(Kaleidowall PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(kaleido_tests tests/core_tests.cpp)
target_link_libraries(kaleido_tests PRIVATE kaleido_core Qt6::Test)
add_test(NAME kaleido_core COMMAND kaleido_tests)
add_executable(kaleido_backend_tests tests/backend_tests.cpp src/mpv_backend.cpp src/mpv_backend.h)
target_include_directories(kaleido_backend_tests PRIVATE src .deps/mpv/include)
target_link_libraries(kaleido_backend_tests PRIVATE Qt6::Core Qt6::Test)
add_test(NAME kaleido_backend COMMAND kaleido_backend_tests)
add_executable(kaleido_export_tests tests/export_tests.cpp src/export_dialog.cpp src/export_dialog.h)
target_link_libraries(kaleido_export_tests PRIVATE kaleido_export Qt6::Widgets Qt6::Test)
if(MSVC)
  target_compile_options(kaleido_export_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME kaleido_export COMMAND kaleido_export_tests)

add_executable(kaleido_preset_tests tests/preset_tests.cpp src/window.cpp src/window_export.cpp
  src/export_dialog.cpp src/canvas.cpp src/mpv_backend.cpp)
target_include_directories(kaleido_preset_tests PRIVATE src .deps/mpv/include)
target_link_libraries(kaleido_preset_tests PRIVATE kaleido_export Qt6::Widgets Qt6::OpenGLWidgets Qt6::Test)
if(MSVC)
  target_compile_options(kaleido_preset_tests PRIVATE /W4 /utf-8)
endif()
add_test(NAME kaleido_presets COMMAND kaleido_preset_tests)
